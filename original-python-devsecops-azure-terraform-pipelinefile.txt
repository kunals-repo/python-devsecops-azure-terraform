name: Python DevSecOps Pipeline

on:
  push:
  workflow_dispatch:

permissions:
  contents: read
  security-events: write

jobs:

  ci:
    name: CI Pipeline
    runs-on: ubuntu-latest

    strategy:
      matrix:
        python-version: [3.11]

    steps:

      - name: Checkout Repository
        uses: actions/checkout@v4

      - name: Setup Python
        uses: actions/setup-python@v5
        with:
          python-version: ${{ matrix.python-version }}

      - name: Cache pip dependencies
        uses: actions/cache@v4
        with:
          path: ~/.cache/pip
          key: ${{ runner.os }}-pip-${{ matrix.python-version }}-${{ hashFiles('requirements.txt') }}

          restore-keys: |
            ${{ runner.os }}-pip-

      - name: Install Dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

      - name: Run Flake8
        run: flake8 app/main.py

      - name: Run Unit Tests
        run: PYTHONPATH=. pytest tests/test_main.py

      - name: Build Python Package
        run: python -m build

      - name: Upload Build Artifact
        uses: actions/upload-artifact@v4
        with:     
          name: python-package-${{ matrix.python-version }}
          path: dist/

# =========================================================
# SECURITY SCAN
# =========================================================

  security_scan:
    name: Trivy Security Scan
    runs-on: ubuntu-latest

    needs: ci

    steps:

      - name: Checkout Repository
        uses: actions/checkout@v4

      - name: Run Trivy Vulnerability Scan
        uses: aquasecurity/trivy-action@v0.35.0

        with:
          scan-type: 'fs'
          scan-ref: '.'

          format: 'sarif'
          output: 'vuln-result.sarif'

          severity: 'CRITICAL,HIGH'
          exit-code: '1'

      - name: Upload Trivy Scan Results to GitHub Security Tab
        uses: github/codeql-action/upload-sarif@v3

        with:
          sarif_file: 'vuln-result.sarif'

# =========================================================
# TERRAFORM INFRASTRUCTURE
# =========================================================

  terraform:
    name: Terraform Infrastructure
    runs-on: ubuntu-latest

    needs: security_scan

    defaults:
      run:
        working-directory: Terraform/


    steps:

      - name: Checkout code
        uses: actions/checkout@v4

      - name: Setup Terraform
        uses: hashicorp/setup-terraform@v3

        with:
          terraform_version: 1.14.5

      - name: Azure Login
        uses: azure/login@v2

        with:
          creds: |
            {
              "clientId": "${{ secrets.AZURE_CLIENT_ID }}",
              "clientSecret": "${{ secrets.AZURE_CLIENT_SECRET }}",
              "subscriptionId": "${{ secrets.AZURE_SUBSCRIPTION_ID }}",
              "tenantId": "${{ secrets.AZURE_TENANT_ID }}"
            }

      - name: Terraform Init
        run: terraform init

      - name: Terraform Validate
        run: terraform validate

      - name: Terraform Plan
        run: terraform plan -out=tfplan

      - name: Terraform Apply
        run: terraform apply -auto-approve tfplan

      - name: Get Terraform Outputs
        id: terraform_outputs

        run: |
          echo "vm_host=$(terraform output -raw vm_public_ip)" >> $GITHUB_OUTPUT
          echo "vm_username=$(terraform output -raw vm_username)" >> $GITHUB_OUTPUT

       outputs:
         vm_host: ${{ steps.terraform_outputs.outputs.vm_host }}
         vm_username: ${{ steps.terraform_outputs.outputs.vm_username }}

# =========================================================
# APPLICATION DEPLOYMENT
# =========================================================

  deploy:
    name: Deploy Application
    runs-on: ubuntu-latest

    needs: terraform

    steps:

      - name: Checkout Repository
        uses: actions/checkout@v4

      - name: Deploy to Azure Linux VM
        uses: appleboy/ssh-action@v1.0.3

        with:
          host: ${{ needs.terraform.outputs.vm_host }}
          username: ${{ needs.terraform.outputs.vm_username }}
          key: ${{ secrets.VM_SSH_KEY }}

          script: |
            echo "Starting deployment..."

            if [ ! -d "/opt/python-devops-project" ]; then
               sudo mkdir /opt/python-devops-project
            fi

            sudo git clone https://github.com/${{ github.repository }} /opt/python-devops-project

            sudo chown -R $USER:$USER /opt/python-devops-project

            cd /opt/python-devops-project

            sudo git pull origin main

            sudo python3 -m venv venv

            source venv/bin/activate

            python3 -m pip install --upgrade pip

            pip install -r requirements.txt

            sudo tee /etc/systemd/system/flaskapp.service > /dev/null <<EOF
            [Unit]
            Description=Flask Application
            After=network.target

            [Service]
            User=${{ needs.terraform.outputs.vm_username }}
            WorkingDirectory=/opt/python-devops-project

            ExecStart=/opt/python-devops-project/venv/bin/gunicorn \ --bind 0.0.0.0:5000 \ app.main:app

            Restart=always

            [Install]
            WantedBy=multi-user.target
            EOF

            sudo systemctl daemon-reload

            sudo systemctl enable flaskapp

            sudo systemctl restart flaskapp

            echo "Deployment Successful!"


# =========================================================
# when we running jobs on self-hosted runner then , we need to install the docker package on it.
# installing the docker inside the self-hosted runner , because appleboy/ssh-action@v1.0.3 github actions needs the docker package on linux vm.
# So actions requires the container to run. Also , add the linux user into the docker group. So user can run required docker command
